
using System;
using System.Threading.Tasks;
using FrameWork.NetWork.V4;
using Microsoft.Extensions.DependencyInjection;

namespace TestNamespace
{
    partial class GameHandler
    {
        public sealed class Dispatcher : IPacketDispatcher<GameHandler>
        {
            public void Dispatch(
                GameHandler handler,
                byte opcode,
                ReadOnlyMemory<byte> payload,
                IServiceProvider services,
                IPacketSerializer serializer,
                IConnectionContext connection)
            {
                switch (opcode)
                {
                    case 0x01:
                    {
                        handler.HandlePing();
                        break;
                    }
                    case 0x10:
                    {
                        var request = serializer.Deserialize<TestNamespace.LoginRequest>(payload.Span);
                        var response = handler.HandleLogin(request);
                        if (response != null)
                            connection.SendResponse(0x11, response);
                        break;
                    }
                    case 0x20:
                    {
                        var request = serializer.Deserialize<TestNamespace.PingRequest>(payload.Span);
                        _ = DispatchAsync_HandlePingAsync(handler, request, connection);
                        break;
                    }
                    case 0x30:
                    {
                        var request = serializer.Deserialize<TestNamespace.NotifyRequest>(payload.Span);
                        _ = DispatchAsync_HandleNotifyAsync(handler, request, connection);
                        break;
                    }
                    case 0x40:
                    {
                        var __scope = services.CreateScope();
                        var __svc_svc = __scope.ServiceProvider.GetRequiredService<TestNamespace.MyService>();
                        _ = DispatchAsync_HandleWithServices(handler, __svc_svc, connection, __scope);
                        break;
                    }
                    default:
                        break;
                }
            }

            private static async Task DispatchAsync_HandlePingAsync(GameHandler handler, TestNamespace.PingRequest request, IConnectionContext connection)
            {
                try
                {
                    var response = await handler.HandlePingAsync(request);
                    if (response != null)
                        connection.SendResponse(0x21, response);
                }
                catch (Exception ex)
                {
                    connection.OnDispatchError(0x20, ex);
                }
            }

            private static async Task DispatchAsync_HandleNotifyAsync(GameHandler handler, TestNamespace.NotifyRequest request, IConnectionContext connection)
            {
                try
                {
                    await handler.HandleNotifyAsync(request);
                }
                catch (Exception ex)
                {
                    connection.OnDispatchError(0x30, ex);
                }
            }

            private static async Task DispatchAsync_HandleWithServices(GameHandler handler, TestNamespace.MyService __svc_svc, IConnectionContext connection, IServiceScope __scope)
            {
                try
                {
                    var response = await handler.HandleWithServices(__svc_svc, connection);
                    if (response != null)
                        connection.SendResponse(0x41, response);
                }
                catch (Exception ex)
                {
                    connection.OnDispatchError(0x40, ex);
                }
                finally
                {
                    __scope.Dispose();
                }
            }
        }
    }
}
